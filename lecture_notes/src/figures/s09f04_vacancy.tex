% Session 9 -- a vacancy as a boundary of zero size, and why that makes it a
% topological diagnostic.
% Placement: sec. "Impurities as diagnostics III", after the bulk-boundary paragraph.
%
% LAYOUT NOTE: entirely plain TikZ, no pgfplots axis, so every coordinate is in
% canvas units throughout and nothing can drift.
\begin{aqmfigure}{fig:s09:vacancy}{Why a single missing atom detects topology. Top: cut a large hole
into a topologically non-trivial insulator or superconductor and bulk-boundary correspondence
guarantees gapless modes running around its edge. Nothing in that argument refers to the hole's
size, so shrinking it down does not switch it off --- a vacancy is a boundary of zero size, and it
binds an in-gap mode. Bottom: this is exactly the diagnostic that a clean spectrum cannot give. A
trivial and a topological insulator with the same gap have bulk spectra that look identical, so
nothing there distinguishes them (left). Add a vacancy and only the topological one develops a state
inside the gap (right). The rule is as simple as it looks: vacancies produce in-gap modes if and
only if the bulk invariant is non-zero, provided the vacancy itself respects the symmetry protecting
that invariant.}

\begin{tikzpicture}[x=1cm, y=1cm]

\def\d{0.245}   % lattice spacing

\begin{scope}[shift={(3.05,0)}]

        \fill[aqmmuted] (\i*\d,\j*\d) circle (1.05pt);
      \fi
    }
  }
  % two arcs rather than one circle, so the chirality arrows are unmissable

  \draw[aqm, draw=aqmaccent, line width=1.5pt, -{Stealth[length=6pt,width=5pt]}]
    (-0.60,0) arc (180:348:0.60);
  \node[aqmlabsmall, anchor=south, align=center, text=aqmaccent] at (0,1.12)
    {gapless mode around a hole};
  \node[aqmlabsmall, anchor=north, align=center, text=aqmmuted] at (0,-1.10)
    {guaranteed by\\[-1pt]bulk-boundary correspondence};
\end{scope}

% ===================== the shrinking arrow ==================================

% ===================== top right: a single vacancy ==========================
\begin{scope}[shift={(8.95,0)}]
  \foreach \i in {-4,...,4} {
    \foreach \j in {-4,...,4} {
      \pgfmathsetmacro{\rr}{abs(\i)+abs(\j)}

      \fi
    }
  }

  \fill[aqmfill] (0,0) circle (0.38);
  \draw[aqm, draw=aqmaccent, line width=1.0pt] (0,0) circle (0.38);
  \node[circle, draw=aqmaccent, fill=white, line width=\aqmlinewidth,
        minimum size=5.5pt, inner sep=0pt] at (0,0) {};
  \node[aqmlabsmall, anchor=south, align=center, text=aqmaccent] at (0,1.12)
    {a bound state on the vacancy};
  \node[aqmlabsmall, anchor=north, align=center, text=aqmmuted] at (0,-1.10)
    {a vacancy is a\\[-1pt]boundary of zero size};
\end{scope}
